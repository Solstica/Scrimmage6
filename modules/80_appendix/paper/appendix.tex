        lambda row: int(np.sum(

\subsection*{A.4 问题四：区域 Benders 与完整域列生成}

def price_all(pool, data, gpu_row, it_row, rdual, cdual,
              cuts, eqdual, tol):
    """完整域精确定价：每个 task-region 的所有合法整数 start 都参与扫描。"""
    cut_signal = np.zeros((R, H_TASK), dtype=float)
    for dual, cut in zip(cdual, cuts):
        cut_signal += -dual * cut.lam[:, :H_TASK]
    profiles = {}
    for task_type, duration in set(zip(data.type_idx.tolist(), data.duration.tolist())):
        prof = overlap_profile(duration)
        for r in range(R):
            example = int(np.flatnonzero(data.type_idx == task_type)[0])
            per_hour = np.zeros(H_TASK)
            for h in range(H_TASK):
                gr, ir = int(gpu_row[r, h]), int(it_row[r, h])
                if gr >= 0:
                    per_hour[h] += -rdual[gr]
                if ir >= 0:
                    per_hour[h] += -rdual[ir] * data.alpha[example]
                per_hour[h] += cut_signal[r, h] * data.pue[r] * data.alpha[example]
            profiles[(task_type, r, int(round(duration * 60)))] = \
                np.convolve(per_hour, prof[::-1], mode="valid")

    active = {}
    for task, region, start in zip(*pool.arrays()):
        active.setdefault((int(task), int(region)), set()).add(int(start))
    additions = []
    for i in range(len(data.task)):
        start0 = int(data.arrival[i])
        end = start0 if data.task.iloc[i].TaskType == "RealTimeInference" \
            else legal_last_start(data, i)
        best = None
        for r in np.flatnonzero(data.legal[i]):
            scores = profiles[(int(data.type_idx[i]), int(r),
                               int(round(data.duration[i] * 60)))][start0:end + 1]
            used = {s for s in active.get((i, int(r)), set()) if start0 <= s <= end}
            if len(used) >= len(scores):
                continue
            local = int(np.argmin(scores))
            s = start0 + local
            if s in used:
                count = min(len(scores), len(used) + 1)
                candidates = np.argpartition(scores, count - 1)[:count]
                local = min((int(k) for k in candidates
                             if start0 + int(k) not in used), key=lambda k: scores[k])
                s = start0 + local
            score = float(scores[local] * data.gpu[i])
            if best is None or score < best[0]:
                best = (score, int(r), s)
        if best is not None and best[0] - eqdual[i] < -tol:
            additions.append((i, best[1], best[2], best[0] - eqdual[i]))
    return sorted(additions, key=lambda item: item[3])
