已有工作负载研究说明，Poisson 近似应由具体时间尺度上的离散度与相关性诊断决定\cite{zou_poissonity_2022}；分层需求预测则强调在细粒度结构与参数稳定性之间进行数据驱动聚合\cite{cohen_dac_2022}。本文据此保留训练段识别出的联合业务标记和类型条件资源标记。

\caption{两阶段短期概率预测与基础调度}

\KwIn{$\mathcal D_{0:2399}$，$\mathcal J_{2376:2399}$，容量与 SLA，$M=10^4$}
\KwOut{24 h 概率预测、实际任务排程与约束审计}

\tcp*[l]{Stage I: structure identification and independent validation}
$(\mathcal D_{tr},\mathcal D_{val},\mathcal D_{te})\leftarrow(0{:}2351,2352{:}2375,2376{:}2399)$\;
$\mathcal M\leftarrow\Diag(\mathcal D_{tr})$ \tcp*[r]{Fano/ACF, Region--TaskType, $(G,D)\mid K$}

Freeze $\mathcal M$ after validation; do not use $\mathcal D_{te}$ for model selection\;

\tcp*[l]{Stage II: refit, final forecast and baseline scheduling}
$\theta_2\leftarrow\Fit(\mathcal M,\mathcal D_{0:2375})$, \quad $\mathcal I_{te}\leftarrow\MC(\theta_2,M)$\;
Initialize hourly GPU/IT/Facility occupancy with unfinished pre-window tasks\;
\ForEach{$j\in\mathcal J_{2376:2399}$ in arrival order}{
  $C_j\leftarrow\{(r,s):\mathrm{SLA},\mathrm{LatestFinish},s+p_j\le2406\}$; if RT, set $s=a_j$\;
  $(r_j^*,s_j^*)\leftarrow\Place(C_j)$ subject to GPU/IT/Facility capacities\;
  \If{$(r_j^*,s_j^*)=\varnothing$}{\KwRet infeasible schedule\;}
  Update occupancy on $[s_j^*,s_j^*+p_j)$\;
}
Audit uniqueness, RT immediacy, capacities, SLA, LatestFinish and terminal boundary\;
