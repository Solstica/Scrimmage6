% 算法 1: 结构化标记复合 Poisson 分层预测与区间估计
\begin{algorithm}[H]
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwInit}{Initialize}
\SetKwInput{KwOut}{Output}
\SetKwFor{For}{for}{do}{end for}
\SetKwFor{ForEach}{for each}{do}{end for}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\SetKw{KwRet}{return}
\caption{结构化标记复合 Poisson 分层预测与区间估计}
\label{alg:q1-prediction}
\KwIn{训练段任务记录 $\mathcal D_{\text{train}}$（0--2351\,h，$T_{\text{train}}=2352$ 小时），待预测时段 $\mathcal{T}=\{2376,\dots,2399\}$，Monte Carlo 模拟次数 $M=10^4$，随机种子 $\text{seed}=20260816$}
\KwInit{固定随机种子，加载训练段数据 $\mathcal D_{\text{train}}$}
\BlankLine
\emph{阶段一：参数估计}\;
$\hat\lambda \leftarrow \frac{1}{T}\sum_{t=0}^{T-1} n_t$，其中 $T=2352$ 为训练小时数\;
$\hat\pi_{rk} \leftarrow c_{rk} / \sum_{r,k} c_{rk}$，其中 $c_{rk}$ 为 $(R=r,K=k)$ 的任务计数\;
\ForEach{$k \in \{\text{RT}, \text{BI}, \text{AT}\}$}{
    从训练段 $k$ 类任务的 $(G,D)$ 样本构造联合经验分布 $\hat F_{GD}^{(k)}$\;
    计算 $\mathbb E[G\mid K=k]$ 和 $\mathbb E[G^2\mid K=k]$\;
}
\BlankLine
\emph{阶段二：解析均值预测}\;
\ForEach{$t \in \mathcal T$}{
    \ForEach{$(r,k) \in \{1,\dots,6\}\times\{\text{RT},\text{BI},\text{AT}\}$}{
        计算 $\hat Y_{rkt}\leftarrow\hat\lambda\hat\pi_{rk}\mathbb E[G\mid K=k]$\;
    }
    向上聚合得到 Region、TaskType 和 System 层预测\;
}
\BlankLine
\emph{阶段三：Monte Carlo 预测区间}\;
\For{$m\leftarrow1$ \KwTo $M$}{
    \ForEach{$t\in\mathcal T$}{
        从 $\mathrm{Poisson}(\hat\lambda)$ 抽取小时任务数 $n$\;
        \For{$i\leftarrow1$ \KwTo $n$}{
            从 $\hat\pi_{rk}$ 抽取业务标记 $(r_i,k_i)$，再从 $\hat F_{GD}^{(k_i)}$ 抽取资源标记 $(g_i,d_i)$\;
            累加得到 $Y_{r_i k_i t}^{(m)}$\;
        }
    }
}
取模拟分布的 5\% 和 95\% 分位数作为 90\% 预测区间\;
\KwOut{四层预测均值、90\% 预测区间及 MAE、RMSE、WAPE、PICP、MPIW}
\end{algorithm}

% 算法 2: 基础算力调度
\begin{algorithm}[H]
\setstretch{0.92}
\SetAlFnt{\normalsize}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{Input}
\SetKwInput{KwInit}{Initialize}
\SetKwInput{KwOut}{Output}
\SetKwFor{For}{for}{do}{end for}
\SetKwFor{ForEach}{for each}{do}{end for}
\SetKwIF{If}{ElseIf}{Else}{if}{then}{else if}{else}{end if}
\caption{本地优先最小调整基础算力调度}
\label{alg:q1-scheduling}
\KwIn{2376--2399\,h 实际到达任务集 $\mathcal J$；各区域 GPU、IT、Facility 容量；任务类型功率映射；2376\,h 前历史任务在收尾时域中的固定占用}
\KwInit{重构 2376--2405\,h 的 GPU/IT/Facility 初始状态}
\BlankLine
\ForEach{$t\in\{2376,\dots,2399\}$}{
    \ForEach{时刻 $t$ 到达的实时推理任务 $j$}{
        固定 $s_j=t$；先检查来源区域，若不可行则按网络时延升序枚举全部 20 ms SLA 合法区域\;
        选择首个同时满足 GPU、IT、Facility、LatestFinish 与 2406 边界的区域并更新状态\;
    }
    \ForEach{时刻 $t$ 到达的批量推理任务 $j$}{
        先在来源区域完整合法时间域内寻找最早可行开工时刻\;
        若来源区域全域不可行，再按网络时延升序枚举其他 80 ms SLA 合法区域\;
        记录最早可行方案并更新状态\;
    }
    \ForEach{时刻 $t$ 到达的 AI 训练任务 $j$}{
        先在来源区域完整合法时间域内寻找最早可行开工时刻\;
        若来源区域全域不可行，再枚举其他 150 ms SLA 合法区域\;
        记录最早可行方案并更新状态\;
    }
}
逐任务检查不可拆分、不可抢占、LatestFinish 与 finish$\le2406$；逐区域小时检查 GPU、IT、Facility 容量和 SLA\;
\KwOut{任务执行区域与开工/完成时刻、甘特图数据、区域 GPU 利用率和硬约束审计结果}
\end{algorithm}
