
\begin{algorithm}[H]
\SetAlgoLined
\setstretch{0.82}
\SetAlFnt{\small}
\SetArgSty{textnormal}
\SetKwInput{KwIn}{输入}
\SetKwInput{KwInit}{初始化}
\SetKwInput{KwOut}{输出}
\SetKwFor{For}{循环}{执行}{结束循环}
\SetKwFor{ForEach}{对每个}{执行}{结束}
\SetKwIF{If}{ElseIf}{Else}{若}{则}{否则若}{否则}{结束判断}
\caption{SFETA：完整合法域动态边际任务分配}
\label{alg:q2-sfeta}
\KwIn{任务集 $\mathcal I$，能源基准，网络与资源边界，目标模式 $m$，收敛容差 $\epsilon$}
\KwInit{以“来源区域、到达即执行”构造初始排程，首轮允许消除容量超限}
\ForEach{任务 $i$}{构造区域可行域 $R_i$、时间可行域 $T_i$ 与完整合法域 $C_i=R_i\times T_i$\;}
按 $\kappa_i=(|R_i|,\sigma_i,-g_ip_i,\mathrm{ID}_i)$ 的字典序确定扫描顺序\;
\For{$k\leftarrow1$ \KwTo $K_{\max}$}{
  重构当前 GPU、IT 功率、设施功率与逐时能源状态\;
  \ForEach{排序后的任务 $i$}{
    暂时移除任务 $i$，得到剩余排程\;
    枚举 $(r,s)\in C_i$，检查资源容量、网络时延、最晚完成时间与 2406 h 终止边界；实时推理任务另要求 $s=a_i$\;
    对可行候选计算 $\Delta\mathrm{GridPurchase}$、$\Delta\mathrm{Curtailment}$、$\Delta\mathrm{Cost}$ 与 $\Delta\mathrm{Carbon}$\;
    \If{$m$ 为成本优先}{先比较运行成本，再比较碳排放和弃电变化；同值时优先较早开工、本地区域和较低网络时延\;}
    \Else{先比较碳排放，再比较运行成本和弃电变化；同值时优先较早开工、本地区域和较低网络时延\;}
    将任务插入最优候选位置；若不存在可行新位置，则保留仍满足约束的原位置，否则判为不可行\;
  }
  重新计算目标值并检查全部约束\;
  \If{$k\ge2$ 且相对改善 $<\epsilon$ 且排程不再变化}{停止循环\;}
}
计算运行成本、碳排放、新能源利用率、迁移率、等待时间和网络时延\;
\KwOut{任务执行区域、开工时刻、评价指标与约束检验结果}
\end{algorithm}
